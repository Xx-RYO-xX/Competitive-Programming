## 10進数からn進数
### コード

```python
def base_n(number: int, base: int) -> str:
    """
    10進数をn進数に変換する関数だっピ\n
    ￣￣V￣￣￣￣￣￣￣￣￣￣￣￣￣￣\n
    ( ・3・)\n
    /｜｜｜\\

    Parameters
    ----------
    number : int
        10進数の数値
    base : int
        変換したいn進数（2〜36の範囲を想定）

    Returns
    -------
    return : str
        n進数に変換された文字列
    """
    if base < 2:
        raise ValueError("baseは2以上の整数を指定してほしいっピ！")

    digits = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"

    if base > len(digits):
        raise ValueError(f"この関数は{len(digits)}進数までしか対応してないっピ！")

    if number < 0:
        return "-" + base_n(-number, base)
    elif number < base:
        return digits[number]
    else:
        return base_n(number // base, base) + digits[number % base]
```
## n進数から10進数
### コード
```python
def base_10(number_str: str, base: int) -> int:
    """
    n進数を10進数に変換する関数だっピ\n
    ￣￣V￣￣￣￣￣￣￣￣\n
    ( ・3・)\n
    /｜｜｜\\
    
    Parameters
    ----------
    number_str : str
        n進数の文字列
    base : int
        変換元のn進数（2〜62を想定）
        
    Returns
    -------
    return : int
        10進数に変換された数値
    """
    if base < 2:
        raise ValueError("baseは2以上を指定してほしいっピ！")

    digits = '0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ'

    if base > len(digits):
        raise ValueError(f"Base {base} is not supported. Maximum base is {len(digits)}.")

    is_negative = False
    if number_str.startswith('-'):
        is_negative = True
        number_str = number_str[1:] 
        
    if not number_str:
        raise ValueError("文字列が空だっピ！")

    value = 0
    for char in number_str:
        if base <= 36:
            char = char.lower()
            
        if char not in digits:
            raise ValueError(f"不正な文字 '{char}' が含まれているっピ！")
            
        char_value = digits.index(char)
        
        if char_value >= base:
            raise ValueError(f"文字 '{char}' は{base}進数では使えないっピ！")
            
        value = value * base + char_value
        
    return -value if is_negative else value
```
